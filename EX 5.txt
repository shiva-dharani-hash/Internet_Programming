        private void printAnswer(PrintWriter out,
                             int number,
                             String answer,
                             String correctAnswer) {
        boolean correct = correctAnswer.equals(answer);
        String cssClass;
        if (correct) {
            cssClass = "correct";
        } else {
            cssClass = "wrong";
        }
        String displayAnswer;
        if (answer == null || answer.trim().isEmpty()) {
            displayAnswer = "Not Answered";
        } else {
            displayAnswer = escapeHtml(answer);
        }
       out.println("<div class='answer " + cssClass + "'>");
        out.println("<b>Question " + number + ":</b> ");
        out.println(displayAnswer);
        if (correct) {
            out.println(" ✓ Correct");
        } else {
            out.println(" ✗ Wrong");
            out.println("<br>");
            out.println("<b>Correct Answer:</b> ");
            out.println(escapeHtml(correctAnswer));
        }
        out.println("</div>");
    }
    private String escapeHtml(String text) {

        if (text == null) {
            return "";
        }
        return text
                .replace("&", "&amp;")
                .replace("<", "&lt;")
                .replace(">", "&gt;")
                .replace("\"", "&quot;")
                .replace("'", "&#39;");
    }
